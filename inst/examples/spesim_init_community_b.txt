# =====================================================================
# spesim_init_community_b.txt — Community B: highly uneven
# ---------------------------------------------------------------------
# 20 species: one dominant holds 81% of all individuals; the remaining
# 19 species share 19% equally (1% each).
# No environmental filtering, no spatial clustering, no interactions.
# This community represents extreme unevenness typical of systems
# shaped by competitive dominance, disturbance, or stress gradients.
# =====================================================================

# --- Reproducibility --------------------------------------------------
SEED = 42

# --- Output prefix ----------------------------------------------------
OUTPUT_PREFIX = out/community_b

# --- Community size and composition ----------------------------------
N_INDIVIDUALS = 2000
N_SPECIES     = 20

# --- Domain type (convex polygon, default) ---------------------------
DOMAIN_TYPE = polygon
LINEAR_AXIS = x
LINEAR_WRAP = FALSE
LINEAR_JITTER_SD = 0.0
DISTANCE_METRIC = auto
ENV_COVARIATES_FILE = ""
ENV_DRIVERS = temperature, elevation, rainfall

# --- High-level model family -----------------------------------------
MODEL_FAMILY = manual

# --- SAD: custom dominant + equal-rare vector -------------------------
# Species A (the first label) gets 81% of individuals.
# The remaining 19 species (B through T) each get 1%.
# c(0.81, rep(0.01, 19)) sums to exactly 1.0 and is treated as
# species probabilities by generate_sad().
SAD_MODEL  = custom
SAD_VECTOR = c(0.81, rep(0.01, 19))

# --- No environmental filtering ---------------------------------------

# --- No local interactions --------------------------------------------
INTERACTION_RADIUS = 0

# --- Spatial point processes ------------------------------------------
# Poisson (CSR) for both species A and the rare species.
# The extreme unevenness comes entirely from the SAD, not from any
# spatial structuring — this isolates the SAD effect cleanly.
SPATIAL_PROCESS_A      = poisson
SPATIAL_PROCESS_OTHERS = poisson

# --- Quadrat sampling -------------------------------------------------
SAMPLING_SCHEME     = random
N_QUADRATS          = 20
QUADRAT_SIZE_OPTION = medium

# --- Advanced analysis panel ------------------------------------------
ADVANCED_ANALYSIS = FALSE
